    <script>
        // ⚠️ PEGA AQUÍ TU URL DE GOOGLE APPS SCRIPT
        const SCRIPT_URL = "https://script.google.com/macros/s/AKfycbxU3Y2D38LCC_Km36hiGCsmL86AP6F5WJmmOM6aRWaCz0qf62brHWaiPRTMyVAEmfCU/exec";

        let currentStep = 1;
        const totalSteps = 5;

        function startSurvey() {
            document.getElementById('welcome-card').classList.add('hidden');
            document.getElementById('survey-form').classList.remove('hidden');
            updateSteps();
        }

        function changeStep(direction) {
            const steps = document.querySelectorAll('.form-step');
            
            // Validar campos requeridos del paso actual antes de avanzar
            if (direction === 1) {
                const currentBlock = document.querySelector(`.form-step[data-step="${currentStep}"]`);
                const inputs = currentBlock.querySelectorAll('input[required], select[required]');
                let valid = true;
                inputs.forEach(input => {
                    if (input.type === 'radio') {
                        const group = currentBlock.querySelectorAll(`input[name="${input.name}"]`);
                        const checked = Array.from(group).some(r => r.checked);
                        if (!checked) valid = false;
                    } else if (!input.value) {
                        valid = false;
                    }
                });
                if (!valid) {
                    alert("Por favor,complete las preguntas obligatorias de este apartado antes de continuar.");
                    return;
                }
            }

            currentStep += direction;
            if (currentStep < 1) currentStep = 1;
            if (currentStep > totalSteps) currentStep = totalSteps;
            updateSteps();
        }

        function updateSteps() {
            const steps = document.querySelectorAll('.form-step');
            steps.forEach(step => {
                if (parseInt(step.dataset.step) === currentStep) {
                    step.classList.remove('hidden');
                } else {
                    step.classList.add('hidden');
                }
            });

            // Actualizar barra de progreso
            const progress = (currentStep / totalSteps) * 100;
            document.getElementById('progress-bar').style.width = progress + '%';

            // Botones
            document.getElementById('prev-btn').classList.toggle('hidden', currentStep === 1);
            document.getElementById('next-btn').classList.toggle('hidden', currentStep === totalSteps);
            document.getElementById('submit-btn').classList.toggle('hidden', currentStep !== totalSteps);
            
            window.scrollTo({ top: 0, behavior: 'smooth' });
        }

        async function submitForm(e) {
            e.preventDefault();
            const submitBtn = document.getElementById('submit-btn');
            submitBtn.disabled = true;
            submitBtn.innerText = "Enviando respuestas...";

            const formData = new FormData(e.target);
            const data = { tipo_entrada: "Web" };
            formData.forEach((value, key) => { data[key] = value; });

            try {
                const response = await fetch(SCRIPT_URL, {
                    method: 'POST',
                    body: JSON.stringify(data)
                });
                const result = await response.json();
                
                if (result.status === "success") {
                    document.getElementById('survey-form').classList.add('hidden');
                    document.getElementById('success-card').classList.remove('hidden');
                } else {
                    alert("Error al guardar: " + result.message);
                    submitBtn.disabled = false;
                    submitBtn.innerText = "Enviar respuestas";
                }
            } catch (error) {
                alert("Error de conexión con el servidor. Verifique su red e intente nuevamente.");
                submitBtn.disabled = false;
                submitBtn.innerText = "Enviar respuestas";
            }
        }
    </script>
