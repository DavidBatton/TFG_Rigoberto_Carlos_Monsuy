    <script>
        const SCRIPT_URL = "https://script.google.com/macros/s/AKfycbxU3Y2D38LCC_Km36hiGCsmL86AP6F5WJmmOM6aRWaCz0qf62brHWaiPRTMyVAEmfCU/exec";

        async function submitPaperForm(e) {
            e.preventDefault();
            const btn = document.getElementById('save-paper-btn');
            btn.disabled = true;
            btn.innerText = "Guardando cuestionario...";

            const formData = new FormData(e.target);
            const data = { tipo_entrada: "Papel" };
            formData.forEach((value, key) => { data[key] = value; });

            try {
                const response = await fetch(SCRIPT_URL, {
                    method: 'POST',
                    body: JSON.stringify(data)
                });
                const result = await response.json();
                if (result.status === "success") {
                    alert("Cuestionario en papel guardado correctamente con ID: " + result.id);
                    e.target.reset();
                } else {
                    alert("Error: " + result.message);
                }
            } catch (err) {
                alert("Error de conexión al guardar.");
            } finally {
                btn.disabled = false;
                btn.innerText = "Guardar Cuestionario en Papel";
            }
        }
    </script>
