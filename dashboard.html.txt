    <script>
        const SCRIPT_URL = "https://script.google.com/macros/s/AKfycbxU3Y2D38LCC_Km36hiGCsmL86AP6F5WJmmOM6aRWaCz0qf62brHWaiPRTMyVAEmfCU/exec";
        let globalData = [];
        let charts = {};

        async function loadData() {
            try {
                const response = await fetch(SCRIPT_URL + "?action=getData");
                globalData = await response.json();
                updateDashboard(globalData);
            } catch (err) {
                console.error("Error al cargar datos:", err);
            }
        }

        function updateDashboard(data) {
            document.getElementById('kpi-total').innerText = data.length;
            const webCount = data.filter(d => d.tipo_entrada === "Web").length;
            const paperCount = data.filter(d => d.tipo_entrada === "Papel").length;
            document.getElementById('kpi-web').innerText = webCount;
            document.getElementById('kpi-paper').innerText = paperCount;

            renderChart('chartEdad', 'bar', processField(data, 'b1_edad'), 'Cuestionarios', '#047857');
            renderChart('chartActividad', 'bar', processField(data, 'b1_actividad'), 'Cuestionarios', '#0284c7');
            renderChart('chartExpansion', 'doughnut', processField(data, 'b3_expansion'), 'Cuestionarios', ['#10b981', '#3b82f6', '#f59e0b', '#ef4444', '#8b5cf6']);
            renderChart('chartMaterial', 'bar', processField(data, 'b5_material'), 'Cuestionarios', '#0d9488');
        }

        function processField(data, field) {
            const counts = {};
            data.forEach(row => {
                const val = row[field] || "Sin responder";
                counts[val] = (counts[val] || 0) + 1;
            });
            return {
                labels: Object.keys(counts),
                values: Object.values(counts)
            };
        }

        function renderChart(canvasId, type, statData, label, color) {
            if (charts[canvasId]) charts[canvasId].destroy();
            const ctx = document.getElementById(canvasId).getContext('2d');
            charts[canvasId] = new Chart(ctx, {
                type: type,
                data: {
                    labels: statData.labels,
                    datasets: [{
                        label: label,
                        data: statData.values,
                        backgroundColor: color,
                        borderWidth: 1
                    }]
                },
                options: { responsive: true, maintainAspectRatio: false }
            });
        }

        function exportCSV() {
            if (globalData.length === 0) { alert("No hay datos para exportar."); return; }
            const headers = Object.keys(globalData[0]);
            let csvContent = headers.join(",") + "\n";
            globalData.forEach(row => {
                const values = headers.map(h => `"${(row[h] || "").toString().replace(/"/g, '""')}"`);
                csvContent += values.join(",") + "\n";
            });
            const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
            const link = document.createElement("a");
            link.href = URL.createObjectURL(blob);
            link.download = "oyala_encuestas_jamovi.csv";
            link.click();
        }

        loadData();
